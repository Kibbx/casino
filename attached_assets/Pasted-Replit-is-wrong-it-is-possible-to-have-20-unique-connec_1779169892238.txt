function validateWesternPaylines(paylines) {
  if (!Array.isArray(paylines)) {
    throw new Error("WESTERN_PAYLINES must be an array");
  }

  if (paylines.length !== 20) {
    throw new Error(`Western Slots must have exactly 20 paylines. Found ${paylines.length}`);
  }

  const seen = new Set();

  paylines.forEach((line, index) => {
    const lineNumber = index + 1;

    if (!Array.isArray(line) || line.length !== 5) {
      throw new Error(`Western payline ${lineNumber} must have exactly 5 row entries`);
    }

    line.forEach((row) => {
      if (![0, 1, 2].includes(row)) {
        throw new Error(`Western payline ${lineNumber} has invalid row ${row}`);
      }
    });

    for (let i = 1; i < line.length; i++) {
      const diff = Math.abs(line[i] - line[i - 1]);

      if (diff > 1) {
        throw new Error(
          `Western payline ${lineNumber} is disconnected: ${line.join(",")}`
        );
      }
    }

    const key = line.join(",");

    if (seen.has(key)) {
      throw new Error(`Duplicate Western payline found: ${key}`);
    }

    seen.add(key);
  });

  console.log("Western paylines valid:", paylines.length, "unique connected lines");
}
